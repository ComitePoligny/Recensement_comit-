        }

        function getVal(objet, motsCles) {

            if (!objet) return '';

            const cles = Object.keys(objet);

            for (let mc of motsCles) {

                const cleTrouvee = cles.find(k => k.toLowerCase().trim().replace(/_/g, ' ') === mc.toLowerCase().trim());

                if (cleTrouvee && objet[cleTrouvee]) {

                    return objet[cleTrouvee];

                }

            }

            return '';

        }

        function creerLien(valeur, texte) {

            if (!valeur) return '';

            valeur = String(valeur).trim();

            if (valeur.startsWith('http://') || valeur.startsWith('https://')) {

                return `<a href="${valeur}" target="_blank">${texte || 'Consulter'}</a>`;

            }

            return valeur;

        }
